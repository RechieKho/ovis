    price (decimal, 0)

    give back price of a < price of b

    give back title of a = title of b

    give back title of b + " (" + (price of b as text) + ")"

emma = a book with title "Emma", price 8
books = [dune, emma]
sort books
show first item of books, smallest of books, largest of books
show emma < dune, emma >= dune, emma != dune, dune = a book with title "Dune"
show books contains emma, books contains a book with title "Zed"
show position of emma in books, count of dune in books, position of true in [false, true]
show dune comes before emma, emma comes before dune, dune is the same as dune
show "Book: " + dune, dune as text
show books, "All: " + books, books as text
show smallest of ["pear", "apple"], largest of ["pear", "apple"]

true false true true
true false
1 1 2
false true true
